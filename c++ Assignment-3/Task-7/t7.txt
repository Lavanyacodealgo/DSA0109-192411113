#include <iostream>
#include <cmath>
using namespace std;

class Complex {
    double real, imag;

public:
    Complex(double r = 0, double i = 0) {
        real = r;
        imag = i;
    }

    Complex operator+(Complex c) {
        return Complex(real + c.real, imag + c.imag);
    }

    Complex operator-(Complex c) {
        return Complex(real - c.real, imag - c.imag);
    }

    Complex operator*(Complex c) {
        return Complex(real * c.real - imag * c.imag,
                       real * c.imag + imag * c.real);
    }

    Complex operator/(Complex c) {
        double d = c.real * c.real + c.imag * c.imag;

        if (d == 0) {
            cout << "Division by zero is not possible" << endl;
            return Complex(0, 0);
        }

        return Complex((real * c.real + imag * c.imag) / d,
                       (imag * c.real - real * c.imag) / d);
    }

    bool operator==(Complex c) {
        return real == c.real && imag == c.imag;
    }

    double magnitude() {
        return sqrt(real * real + imag * imag);
    }

    friend ostream& operator<<(ostream& out, Complex c) {
        out << c.real;
        if (c.imag >= 0)
            out << " + " << c.imag << "i";
        else
            out << " - " << -c.imag << "i";
        return out;
    }
};

int main() {
    Complex a(5, 3);
    Complex b(2, 1);

    cout << "First Complex Number: " << a << endl;
    cout << "Second Complex Number: " << b << endl;

    cout << "Addition: " << a + b << endl;
    cout << "Subtraction: " << a - b << endl;
    cout << "Multiplication: " << a * b << endl;
    cout << "Division: " << a / b << endl;

    cout << "Equality: " << (a == b) << endl;
    cout << "Magnitude: " << a.magnitude() << endl;

    double R = 10;
    double L = 0.1;
    double C = 0.0001;
    double f = 50;
    double pi = 3.14159;

    double w = 2 * pi * f;

    Complex resistor(R, 0);
    Complex inductor(0, w * L);
    Complex capacitor(0, -1 / (w * C));

    Complex series = resistor + inductor + capacitor;

    Complex one(1, 0);
    Complex admittance = one / resistor
                       + one / inductor
                       + one / capacitor;

    Complex parallel = one / admittance;

    cout << "Series Impedance: " << series << endl;
    cout << "Parallel Impedance: " << parallel << endl;

    return 0;
}
