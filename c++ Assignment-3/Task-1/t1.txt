
#include <iostream>
#include <string>
using namespace std;

class Book {
    string title;
    string author;
    int copies;

public:
    Book() {
        title = "Unknown";
        author = "Unknown";
        copies = 0;
    }

    Book(string t) {
        title = t;
        author = "Unknown";
        copies = 1;
    }

    Book(string t, string a, int c) {
        title = t;
        author = a;
        copies = c;
    }

    void display() {
        cout << "Title: " << title << endl;
        cout << "Author: " << author << endl;
        cout << "Copies: " << copies << endl;
        cout << endl;
    }
};

int main() {
    Book b1;
    Book b2("C++ Programming");
    Book b3("Java", "James", 5);

    b1.display();
    b2.display();
    b3.display();

    return 0;
}
