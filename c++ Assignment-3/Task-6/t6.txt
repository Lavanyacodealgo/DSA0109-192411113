#include <iostream>
#include <cstring>
using namespace std;

class Text {
    char *str;

public:
    Text(const char *s = "") {
        str = new char[strlen(s) + 1];
        strcpy(str, s);
    }

    Text(const Text &t) {
        str = new char[strlen(t.str) + 1];
        strcpy(str, t.str);
    }

    Text& operator=(const Text &t) {
        if (this != &t) {
            delete[] str;
            str = new char[strlen(t.str) + 1];
            strcpy(str, t.str);
        }
        return *this;
    }

    Text operator+(const Text &t) {
        char *temp = new char[strlen(str) + strlen(t.str) + 1];

        strcpy(temp, str);
        strcat(temp, t.str);

        Text result(temp);
        delete[] temp;

        return result;
    }

    bool operator==(const Text &t) {
        return strcmp(str, t.str) == 0;
    }

    bool operator!=(const Text &t) {
        return strcmp(str, t.str) != 0;
    }

    friend ostream& operator<<(ostream &out, const Text &t) {
        out << t.str;
        return out;
    }

    ~Text() {
        delete[] str;
    }
};

int main() {
    Text a("Hello ");
    Text b("World");
    Text c("!");

    Text d = a + b + c;

    cout << "First Text: " << a << endl;
    cout << "Second Text: " << b << endl;
    cout << "Combined Text: " << d << endl;

    cout << "Equality: " << (a == b) << endl;
    cout << "Inequality: " << (a != b) << endl;

    Text e = a;
    cout << "Copied Text: " << e << endl;

    e = b;
    cout << "Assigned Text: " << e << endl;

    e = e;
    cout << "Self Assignment: " << e << endl;

    return 0;
}
