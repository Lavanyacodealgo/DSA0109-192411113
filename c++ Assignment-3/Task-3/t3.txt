
#include <iostream>
using namespace std;

class Auditorium {
    int rows, seats;
    char** map;

public:
    Auditorium(int r, int s) {
        rows = r;
        seats = s;

        map = new char*[rows];

        for (int i = 0; i < rows; i++) {
            map[i] = new char[seats];

            for (int j = 0; j < seats; j++) {
                map[i][j] = 'A';
            }
        }

        cout << "Auditorium Created" << endl;
    }

    void book(int r, int s) {
        if (r >= 0 && r < rows && s >= 0 && s < seats) {
            if (map[r][s] == 'A') {
                map[r][s] = 'B';
                cout << "Seat Booked" << endl;
            }
            else {
                cout << "Seat Already Booked" << endl;
            }
        }
        else {
            cout << "Invalid Seat" << endl;
        }
    }

    void display() {
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < seats; j++) {
                cout << map[i][j] << " ";
            }
            cout << endl;
        }
    }

    ~Auditorium() {
        for (int i = 0; i < rows; i++) {
            delete[] map[i];
        }

        delete[] map;

        cout << "Auditorium Closed" << endl;
    }
};

int main() {
    Auditorium a(2, 3);

    a.book(0, 1);
    a.book(1, 2);

    cout << "Seat Details:" << endl;
    a.display();

    return 0;
}
